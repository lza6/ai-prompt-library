function validateFileUpload(file: File) {
  // Size check (5MB max)
  const maxSize = 5 * 1024 * 1024
  if (file.size > maxSize) {
    throw new Error('File too large (max 5MB)')
  }

  // Type check
  const allowedTypes = ['image/jpeg', 'image/png', 'image/gif']
  if (!allowedTypes.includes(file.type)) {
    throw new Error('Invalid file type')
  }

  // Extension check
  const allowedExtensions = ['.jpg', '.jpeg', '.png', '.gif']
  const extension = file.name.toLowerCase().match(/\.[^.]+$/)?.[0]
  if (!extension || !allowedExtensions.includes(extension)) {
    throw new Error('Invalid file extension')
  }

  return true
}

function validateFileUpload(file: File) {
  // Boyut kontrolü (5MB max)
  const maxSize = 5 * 1024 * 1024
  if (file.size > maxSize) {
    throw new Error('Dosya çok büyük (max 5MB)')
  }

  // Tip kontrolü
  const allowedTypes = ['image/jpeg', 'image/png', 'image/gif']
  if (!allowedTypes.includes(file.type)) {
    throw new Error('Geçersiz dosya tipi')
  }

  // Uzantı kontrolü
  const allowedExtensions = ['.jpg', '.jpeg', '.png', '.gif']
  const extension = file.name.toLowerCase().match(/\.[^.]+$/)?.[0]
  if (!extension || !allowedExtensions.includes(extension)) {
    throw new Error('Geçersiz dosya uzantısı')
  }

  return true
}

interface DataLoaderProps<T> {
  url: string
  children: (data: T | null, loading: boolean, error: Error | null) => React.ReactNode
}

export function DataLoader<T>({ url, children }: DataLoaderProps<T>) {
  const [data, setData] = useState<T | null>(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<Error | null>(null)

  useEffect(() => {
    fetch(url)
      .then(res => res.json())
      .then(setData)
      .catch(setError)
      .finally(() => setLoading(false))
  }, [url])

  return <>{children(data, loading, error)}</>
}

interface UseQueryOptions<T> {
  onSuccess?: (data: T) => void
  onError?: (error: Error) => void
  enabled?: boolean
}

export function useQuery<T>(
  key: string,
  fetcher: () => Promise<T>,
  options?: UseQueryOptions<T>
) {
  const [data, setData] = useState<T | null>(null)
  const [error, setError] = useState<Error | null>(null)
  const [loading, setLoading] = useState(false)

  const refetch = useCallback(async () => {
    setLoading(true)
    setError(null)

    try {
      const result = await fetcher()
      setData(result)
      options?.onSuccess?.(result)
    } catch (err) {
      const error = err as Error
      setError(error)
      options?.onError?.(error)
    } finally {
      setLoading(false)
    }
  }, [fetcher, options])

  useEffect(() => {
    if (options?.enabled !== false) {
      refetch()
    }
  }, [key, refetch, options?.enabled])

  return { data, error, loading, refetch }
}

export function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value)

  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value)
    }, delay)

    return () => clearTimeout(handler)
  }, [value, delay])

  return debouncedValue
}

interface State {
  markets: Market[]
  selectedMarket: Market | null
  loading: boolean
}

type Action =
  | { type: 'SET_MARKETS'; payload: Market[] }
  | { type: 'SELECT_MARKET'; payload: Market }
  | { type: 'SET_LOADING'; payload: boolean }

function reducer(state: State, action: Action): State {
  switch (action.type) {
    case 'SET_MARKETS':
      return { ...state, markets: action.payload }
    case 'SELECT_MARKET':
      return { ...state, selectedMarket: action.payload }
    case 'SET_LOADING':
      return { ...state, loading: action.payload }
    default:
      return state
  }
}

const MarketContext = createContext<{
  state: State
  dispatch: Dispatch<Action>
} | undefined>(undefined)

export function MarketProvider({ children }: { children: React.ReactNode }) {
  const [state, dispatch] = useReducer(reducer, {
    markets: [],
    selectedMarket: null,
    loading: false
  })

  return (
    <MarketContext.Provider value={{ state, dispatch }}>
      {children}
    </MarketContext.Provider>
  )
}

export function useMarkets() {
  const context = useContext(MarketContext)
  if (!context) throw new Error('useMarkets must be used within MarketProvider')
  return context
}

interface ErrorBoundaryState {
  hasError: boolean
  error: Error | null
}

export class ErrorBoundary extends React.Component<
  { children: React.ReactNode },
  ErrorBoundaryState
> {
  state: ErrorBoundaryState = {
    hasError: false,
    error: null
  }

  static getDerivedStateFromError(error: Error): ErrorBoundaryState {
    return { hasError: true, error }
  }

  componentDidCatch(error: Error, errorInfo: React.ErrorInfo) {
    console.error('Error boundary caught:', error, errorInfo)
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="error-fallback">
          <h2>Bir şeyler yanlış gitti</h2>
          <p>{this.state.error?.message}</p>
          <button onClick={() => this.setState({ hasError: false })}>
            Tekrar dene
          </button>
        </div>
      )
    }

    return this.props.children
  }
}

export function Dropdown({ options, onSelect }: DropdownProps) {
  const [isOpen, setIsOpen] = useState(false)
  const [activeIndex, setActiveIndex] = useState(0)

  const handleKeyDown = (e: React.KeyboardEvent) => {
    switch (e.key) {
      case 'ArrowDown':
        e.preventDefault()
        setActiveIndex(i => Math.min(i + 1, options.length - 1))
        break
      case 'ArrowUp':
        e.preventDefault()
        setActiveIndex(i => Math.max(i - 1, 0))
        break
      case 'Enter':
        e.preventDefault()
        onSelect(options[activeIndex])
        setIsOpen(false)
        break
      case 'Escape':
        setIsOpen(false)
        break
    }
  }

  return (
    <div
      role="combobox"
      aria-expanded={isOpen}
      aria-haspopup="listbox"
      onKeyDown={handleKeyDown}
    >
      {/* Dropdown implementasyonu */}
    </div>
  )
}

export function Modal({ isOpen, onClose, children }: ModalProps) {
  const modalRef = useRef<HTMLDivElement>(null)
  const previousFocusRef = useRef<HTMLElement | null>(null)

  useEffect(() => {
    if (isOpen) {
      // Şu anki focus'lanmış elementi kaydet
      previousFocusRef.current = document.activeElement as HTMLElement

      // Modal'a focus yap
      modalRef.current?.focus()
    } else {
      // Kapatırken focus'u geri yükle
      previousFocusRef.current?.focus()
    }
  }, [isOpen])

  return isOpen ? (
    <div
      ref={modalRef}
      role="dialog"
      aria-modal="true"
      tabIndex={-1}
      onKeyDown={e => e.key === 'Escape' && onClose()}
    >
      {children}
    </div>
  ) : null
}

class CachedMarketRepository implements MarketRepository {
  constructor(
    private baseRepo: MarketRepository,
    private redis: RedisClient
  ) {}

  async findById(id: string): Promise<Market | null> {
    // Önce önbelleği kontrol et
    const cached = await this.redis.get(`market:${id}`)

    if (cached) {
      return JSON.parse(cached)
    }

    // Cache miss - veritabanından getir
    const market = await this.baseRepo.findById(id)

    if (market) {
      // 5 dakika önbellekle
      await this.redis.setex(`market:${id}`, 300, JSON.stringify(market))
    }

    return market
  }

  async invalidateCache(id: string): Promise<void> {
    await this.redis.del(`market:${id}`)
  }
}

指数退避 Tekrar Deneme
async function fetchWithRetry<T>(
  fn: () => Promise<T>,
  maxRetries = 3
): Promise<T> {
  let lastError: Error

  for (let i = 0; i < maxRetries; i++) {
    try {
      return await fn()
    } catch (error) {
      lastError = error as Error

      if (i < maxRetries - 1) {
        // Exponential backoff: 1s, 2s, 4s
        const delay = Math.pow(2, i) * 1000
        await new Promise(resolve => setTimeout(resolve, delay))
      }
    }
  }

  throw lastError!
}

type Permission = 'read' | 'write' | 'delete' | 'admin'

interface User {
  id: string
  role: 'admin' | 'moderator' | 'user'
}

const rolePermissions: Record<User['role'], Permission[]> = {
  admin: ['read', 'write', 'delete', 'admin'],
  moderator: ['read', 'write', 'delete'],
  user: ['read', 'write']
}

export function hasPermission(user: User, permission: Permission): boolean {
  return rolePermissions[user.role].includes(permission)
}

export function requirePermission(permission: Permission) {
  return (handler: (request: Request, user: User) => Promise<Response>) => {
    return async (request: Request) => {
      const user = await requireAuth(request)

      if (!hasPermission(user, permission)) {
        throw new ApiError(403, 'Insufficient permissions')
      }

      return handler(request, user)
    }
  }
}

class RateLimiter {
  private requests = new Map<string, number[]>()

  async checkLimit(
    identifier: string,
    maxRequests: number,
    windowMs: number
  ): Promise<boolean> {
    const now = Date.now()
    const requests = this.requests.get(identifier) || []

    // Pencere dışındaki eski istekleri kaldır
    const recentRequests = requests.filter(time => now - time < windowMs)

    if (recentRequests.length >= maxRequests) {
      return false  // Rate limit aşıldı
    }

    // Mevcut isteği ekle
    recentRequests.push(now)
    this.requests.set(identifier, recentRequests)

    return true
  }
}

class JobQueue<T> {
  private queue: T[] = []
  private processing = false

  async add(job: T): Promise<void> {

        await this.execute(job)
      } catch (error) {

      }
    }

服务器 HTML 和客户端状态之间的水合作用不匹配
路线级渲染决策，例如预渲染、SWR、ISR 或仅客户端部分
针对延迟加载、延迟水合或有效载荷大小进行性能优化
使用 useFetch 、 useAsyncData 或 $fetch 获取页面或组件数据
Nuxt 路由问题可能与路由参数、中间件或 SSR/客户端差异有关。
水合作用安全
保持首次渲染的确定性。不要将 Date.now() 、 Math.random() 、仅限浏览器使用的 API 或存储读取操作直接放入 SSR 渲染的模板状态中。
当服务器无法生成相同的标记时，将仅限浏览器使用的逻辑移到 onMounted() 、 import.meta.client 、 ClientOnly 或 .client.vue 组件之后。
请使用 Nuxt 的 useRoute() 可组合函数，而不是 vue-router 中的那个。
请勿使用 route.fullPath 来驱动 SSR 渲染的标记。URL 片段仅供客户端使用，这可能会导致水合物不匹配。
将 ssr: false 视为真正仅限浏览器区域的逃生通道，而不是不匹配的默认修复方法。
数据获取
对于页面和组件中需要进行 SSR 安全 API 读取的操作，建议优先 await useFetch() 。它会将服务器获取的数据转发到 Nuxt 有效负载中，从而避免在数据填充时进行二次获取。
当获取器不是简单的 $fetch() 调用、需要自定义键或组合多个异步源时，请使用 useAsyncData() 。
为 useAsyncData() 指定一个稳定的键，以便重用缓存并实现可预测的刷新行为。
确保 useAsyncData() 处理程序没有副作用。它们可以在服务端渲染 (SSR) 和数据填充 (hydration) 期间运行。
对于用户触发的写入或仅限客户端的操作，请使用 $fetch() ，而不是应该从 SSR 获取的顶级页面数据。
对于不应阻塞导航的非关键数据，请使用 lazy: true 、 useLazyFetch() 或 useLazyAsyncData() 在 UI 中处理 status === 'pending' 。
仅对 SEO 或首次绘制不需要的数据使用 server: false 。
使用 pick 减小有效载荷尺寸，并且在不需要深度反应时优先选择较浅的有效载荷。
const route = useRoute()

const { data: article, status, error, refresh } = await useAsyncData(

export default defineNuxtConfig({
  routeRules: {

isr ：在支持的平台上进行增量静态再生

按路由组选择路由规则，而不是全局选择。营销页面、产品目录、仪表盘和 API 通常需要不同的策略。

工作原理
模式 1：页面性能
通过浏览器 MCP 测量真实的浏览器指标：

1. Navigate to each target URL
2. Measure Core Web Vitals:
   - LCP (Largest Contentful Paint) — target < 2.5s
   - CLS (Cumulative Layout Shift) — target < 0.1
   - INP (Interaction to Next Paint) — target < 200ms
   - FCP (First Contentful Paint) — target < 1.8s
   - TTFB (Time to First Byte) — target < 800ms
3. Measure resource sizes:
   - Total page weight (target < 1MB)
   - JS bundle size (target < 200KB gzipped)
   - CSS size
   - Image weight
   - Third-party script weight
4. Count network requests
5. Check for render-blocking resources
模式 2：API 性能
基准测试 API 端点：

1. Hit each endpoint 100 times
2. Measure: p50, p95, p99 latency
3. Track: response size, status codes
4. Test under load: 10 concurrent requests

当然最终你要进行代码质量和逻辑已经完整链路的审查检查审核
五轴评估
每次代码审查都会从以下几个方面评估代码：

1. 正确性
这段代码是否实现了它声称的功能？

它是否符合规格或任务要求？
是否处理边界情况（null、空值、边界值）？
是否处理错误路径（而不仅仅是正常路径）？
它是否通过了所有测试？这些测试是否真正测试了正确的内容？
是否存在差一错误、竞争条件或状态不一致？
2. 可读性和简洁性
其他工程师（或代理人）能否在作者不解释的情况下理解这段代码？

名称是否具有描述性且符合项目规范？（不要使用没有上下文的 temp 、 data 、 result 等名称）
控制流程是否简单明了（避免嵌套三元运算符、深度回调）？
代码的组织结构是否符合逻辑（相关代码分组，模块边界清晰）？
有哪些“巧妙”的技巧应该简化？
能否用更少的代码行实现？ （明明 100 行就能搞定，却用了 1000 行，这显然是不合理的）
抽象概念的复杂性是否物有所值？ （在第三个用例之前不要进行概括）
注释是否有助于阐明不明显的意图？（但不要给显而易见的代码添加注释。）
是否存在死代码：无操作变量（ _unused ）、向后兼容的垫片，或 // removed 注释？
3. 建筑
这种改变符合系统设计吗？

它遵循现有模式还是引入了新的模式？如果是新的模式，其合理性如何？
它是否保持了清晰的模块边界？
是否存在应该共享的重复代码？
依赖关系是否朝着正确的方向流动（没有循环依赖）？
抽象级别是否合适（既不过分设计，又不会过于耦合）？
4. 安全性
有关详细的安全指导，请参阅 security-and-hardening 。此更改是否会引入漏洞？

用户输入是否经过验证和清理？
代码、日志和版本控制系统中是否包含机密信息？
是否在需要的地方进行了身份验证/授权检查？
SQL 查询是否参数化（不使用字符串拼接）？
输出是否经过编码以防止跨站脚本攻击？
依赖项是否来自可信来源，且不存在已知漏洞？
来自外部来源（API、日志、用户内容、配置文件）的数据是否被视为不可信数据？
外部数据流在使用于逻辑或渲染之前，是否在系统边界进行验证？
5. 性能
有关详细性能分析和优化，请参阅 performance-optimization 。此更改是否引入了性能问题？

是否存在 N+1 查询模式？
是否存在无限制循环或无限制数据获取？
有哪些同步操作应该改为异步操作？
UI 组件中是否存在不必要的重新渲染？
列表端点是否存在分页缺失的情况？
在热路径中创建了任何大型物体吗？
更改尺寸
小规模、重点突出的变更更容易审核、合并速度更快、部署更安全。请针对以下规模的变更进行调整：

~100 lines changed   → Good. Reviewable in one sitting.
~300 lines changed   → Acceptable if it's a single logical change.
~1000 lines changed  → Too large. Split it.
何谓“一次更改”：指针对单一问题的独立修改，包含相关测试，且提交后系统功能保持正常运行。它是某个功能的一部分，而非整个功能。

当变化幅度过大时，应采取分步策略：

战略	如何	什么时候
堆	提交一个小改动，然后以此为基础开始下一个改动。	顺序依赖关系
按文件组	针对需要不同审阅者的群体，分别进行修改。	跨领域关注点
水平的	首先创建共享代码/存根，然后创建消费者	分层架构
垂直的	将功能拆分成更小的全栈切片。	专题作品
当可以接受大幅更改时： 完全删除文件和自动重构，审阅者只需验证意图，而无需逐行检查。

将重构工作与新功能开发分开。 如果一项更改既重构了现有代码又添加了新行为，则属于两项更改——请分别提交。小的清理工作（例如变量重命名）可以由审核人员酌情决定是否包含在内。

变更说明
每一次变更都需要一个独立于版本控制历史记录的描述。

第一行： 简短、命令式、独立。“删除 FizzBu​​zz RPC”，而不是“正在删除 FizzBu​​zz RPC”。必须提供足够的信息，以便查看历史记录的人无需阅读差异对比即可理解更改内容。

正文： 说明正在发生哪些变化以及原因。请包含代码本身无法直接体现的背景信息、决策和推理过程。在相关情况下，请提供错误编号、基准测试结果或设计文档的链接。如果存在方法上的不足，请予以说明。

反模式： “修复错误”、“修复构建”、“添加补丁”、“将代码从 A 移动到 B”、“阶段 1”、“添加便捷函数”。

审核流程
第一步：了解背景
在查看代码之前，先理解其意图：

- What is this change trying to accomplish?
- What spec or task does it implement?
- What is the expected behavior change?
第二步：先复习测试题。
测试揭示意图和覆盖范围：

- Do tests exist for the change?
- Do they test behavior (not implementation details)?
- Are edge cases covered?
- Do tests have descriptive names?
- Would the tests catch a regression if the code changed?
步骤三：审查实施情况
带着对五个维度的思考，逐行分析代码：

For each file changed:
1. Correctness: Does this code do what the test says it should?
2. Readability: Can I understand this without help?
3. Architecture: Does this fit the system?
4. Security: Any vulnerabilities?
5. Performance: Any bottlenecks?
第四步：对结果进行分类
请为每条评论标注其严重程度，以便作者了解哪些是必填项，哪些是选填项：

前缀	意义	作者行动
（无前缀）	所需变更	必须在合并前解决
批判的：	区块合并	安全漏洞、数据丢失、功能故障
镍：	次要的，可选	作者可以忽略格式和样式偏好
可选： / 考虑：	建议	值得考虑，但并非必须。
供参考	仅供参考	无需采取任何行动——供日后参考
这样可以防止作者将所有反馈都视为强制性的，从而避免在可选项上浪费时间。

步骤五：验证验证结果
请查看作者的核实文章：

- What tests were run?
- Did the build pass?
- Was the change tested manually?
- Are there screenshots for UI changes?
- Is there a before/after comparison?
多模型评论模式
针对不同的评审视角，采用不同的模型：

Model A writes the code
    │
    ▼
Model B reviews for correctness and architecture
    │
    ▼
Model A addresses the feedback
    │
    ▼
Human makes the final call
这样可以发现单个模型可能遗漏的问题——不同的模型有不同的盲点。

审核代理的示例提示：

Review this code change for correctness, security, and adherence to
our project conventions. The spec says [X]. The change should [Y].
Flag any issues as Critical, Important, or Suggestion.
死代码卫生
在任何重构或实现变更之后，请检查是否存在孤立代码：

识别当前无法访问或未使用的代码
明确列出。
删除前请先询问： “我是否应该删除这些现在未使用的元素：[列表]？”
不要留下无用的代码——这会给未来的读者和代理商带来困扰。但也不要默默删除你不确定的代码。如有疑问，请咨询。

DEAD CODE IDENTIFIED:
- formatLegacyDate() in src/utils/date.ts — replaced by formatDate()
- OldTaskCard component in src/components/ — replaced by TaskCard
- LEGACY_API_URL constant in src/config.ts — no remaining references
→ Safe to remove these?
审查速度
缓慢的评审流程会阻碍整个团队的工作。切换工作内容进行评审的成本低于其他人因等待而承受的成本。

请在 1 个工作日内回复 ——这是最长期限，而非目标期限。
理想的节奏： 收到审核请求后应尽快回复，除非正在专注编码。通常情况下，一次变更应该在一天内完成多轮审核。
优先考虑快速的个人反馈， 而非快速的最终批准。即使需要多轮反馈，快速反馈也能减少挫败感。
重大变更： 请作者将变更拆分，而不是审查一个庞大的变更集。
处理分歧
解决评论争议时，请遵循以下层级：

技术事实和数据凌驾于观点和偏好之上。
风格指南是风格方面的绝对权威。
软件设计必须基于工程原理进行评估，而不是基于个人偏好。
如果代码库一致性不会降低整体健康状况，那么它是可以接受的。
不要接受“我稍后会清理”的说法。 经验表明，延迟清理很少能真正落实。除非是真正的紧急情况，否则提交前必须完成清理工作。如果此次更改无法解决相关问题，则要求提交一个包含自我分配的缺陷报告。

评论中的诚实
在审查代码时——无论是您自己、其他代理还是人编写的代码：

不要草率盖章。 “LGTM”而没有经过审核的证据对任何人都没有帮助。
不要淡化真正的问题。 “这可能只是个小问题”，而实际上这是一个会影响生产环境的漏洞，这种说法是不诚实的。
尽可能量化问题。 “这条 N+1 查询将使列表中每个项目增加 50 毫秒的延迟”比“这可能会很慢”要好。
对存在明显问题的方案要提出质疑。 阿谀奉承是评审中的败笔。如果方案实施存在问题，请直接指出并提出替代方案。
优雅地接受修改。 如果作者了解全部上下文并且不同意，请尊重他们的判断。评论代码，而不是人——将人身攻击的批评重新组织，使其聚焦于代码本身。
依赖纪律
代码审查的一部分是依赖关系审查：

添加任何依赖项之前：

现有的技术栈能解决这个问题吗？（通常可以。）
依赖项有多大？（检查软件包影响。）
项目还在积极维护吗？（查看最近一次提交记录和未解决的问题。）
它是否存在已知漏洞？（ npm audit ）
许可证是什么？（必须与项目兼容。）
原则： 优先使用标准库和现有工具，而不是新的依赖项。任何依赖项都是一种负担。

常见理由
合理化	现实
“它能用，这就足够了。”	可读性差、不安全或架构错误的代码会造成债务不断累积。
“是我写的，所以我知道它是正确的。”	作者往往对自己的假设视而不见。任何修改都需要其他人的审阅。
“我们稍后再清理。”	“以后再说”永远不会到来。审核是质量把关——一定要好好利用它。合并前必须进行清理，而不是合并后。
“人工智能生成的代码可能没问题”	人工智能代码需要更严格的审查，而不是放松审查。即使出错，它也显得自信且合情合理。
“测试通过了，所以很好。”	测试是必要的，但并非充分条件。测试无法发现架构问题、安全问题或可读性问题。

仅检查测试是否通过的评审（忽略其他维度）
“LGTM” 缺乏实际审查的证据
未经安全审查的安全敏感变更
过大的 PR，无法进行全面审核（请将其拆分）。
修复 bug 的 PR 没有进行回归测试
评论缺少严重性标签——这使得区分哪些是必需的，哪些是可选的变得模糊不清。
接受“以后再解决”这种说法——永远不会实现。

规格漂移：其成因及预防方法
当人工智能做出未经明确规定的“合理假设”时，就会出现偏差。如果没有规范，人工智能会在不知不觉中做出成千上万个微小的决策——每一个决策都可能偏离预期目标。

预防：

首先要制定宪法 ——项目层面的约束从根本上防止偏离正轨。
在具体说明之前先做表面假设 ——错误的假设会导致错误的应用规范，最终导致错误的代码。
规划前务必澄清 ——spec.md 中未解决的歧义会导致错误的架构。
具体性胜过冗长 ——“资源未找到时返回 404”胜过三段文字。
实施前锁定合同 ——第四阶段期间绝不修改 contracts/
每个任务一个独立的上下文 ——每个任务使用一个全新的 AI 会话，可以消除累积的假设。
提交门 ——只有当前任务通过测试后才能执行下一个任务
有关详细的故障模式，请参阅 references/anti-patterns.md 。
